cmake_minimum_required(VERSION 3.16)
project(rtos_memory_management C)
set(CMAKE_C_STANDARD 11)
add_library(memory_pool src/memory_pool.c)
target_include_directories(memory_pool PUBLIC include)
add_executable(memory_sim src/host_main.c)
target_link_libraries(memory_sim PRIVATE memory_pool)
enable_testing()
add_executable(test_pool tests/test_pool.c)
target_link_libraries(test_pool PRIVATE memory_pool)
add_test(NAME pool_tests COMMAND test_pool)
