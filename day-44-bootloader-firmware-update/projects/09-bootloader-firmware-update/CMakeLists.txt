cmake_minimum_required(VERSION 3.16)
project(bootloader_firmware_update C)
set(CMAKE_C_STANDARD 11)
add_library(bootloader_core src/bootloader.c)
target_include_directories(bootloader_core PUBLIC include)
add_executable(bootloader_sim src/host_main.c)
target_link_libraries(bootloader_sim PRIVATE bootloader_core)
enable_testing()
add_executable(test_bootloader tests/test_bootloader.c)
target_link_libraries(test_bootloader PRIVATE bootloader_core)
add_test(NAME bootloader_tests COMMAND test_bootloader)
