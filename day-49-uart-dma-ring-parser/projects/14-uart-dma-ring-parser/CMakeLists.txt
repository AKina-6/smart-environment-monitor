cmake_minimum_required(VERSION 3.16)
project(uart_dma_ring_parser C)
set(CMAKE_C_STANDARD 11)
add_library(uart_core src/uart_protocol.c)
target_include_directories(uart_core PUBLIC include)
add_executable(uart_sim src/host_main.c)
target_link_libraries(uart_sim PRIVATE uart_core)
enable_testing()
add_executable(test_uart tests/test_uart.c)
target_link_libraries(test_uart PRIVATE uart_core)
add_test(NAME uart_tests COMMAND test_uart)
