cmake_minimum_required(VERSION 3.16)
project(sensor_acquisition_pipeline C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(sensor_core src/sensor/filter.c src/sensor/acquisition.c)
target_include_directories(sensor_core PUBLIC include)
add_library(host_sensor src/platform/host/host_sensor.c)
target_include_directories(host_sensor PUBLIC include src/platform/host)
add_executable(sensor_pipeline_sim src/platform/host/host_main.c)
target_link_libraries(sensor_pipeline_sim PRIVATE sensor_core host_sensor)
enable_testing()
add_executable(test_filter tests/test_filter.c)
target_link_libraries(test_filter PRIVATE sensor_core m)
add_test(NAME filter_tests COMMAND test_filter)
add_executable(test_pipeline tests/test_pipeline.c)
target_link_libraries(test_pipeline PRIVATE sensor_core host_sensor)
add_test(NAME pipeline_tests COMMAND test_pipeline)
