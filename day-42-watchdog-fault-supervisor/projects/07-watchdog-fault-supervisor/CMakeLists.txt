cmake_minimum_required(VERSION 3.16)
project(watchdog_fault_supervisor C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(supervisor_core src/supervisor/supervisor.c src/supervisor/reset_log.c)
target_include_directories(supervisor_core PUBLIC include)
add_executable(watchdog_supervisor_sim src/platform/host/host_main.c)
target_link_libraries(watchdog_supervisor_sim PRIVATE supervisor_core)
enable_testing()
add_executable(test_supervisor tests/test_supervisor.c)
target_link_libraries(test_supervisor PRIVATE supervisor_core)
add_test(NAME supervisor_tests COMMAND test_supervisor)
add_executable(test_reset_log tests/test_reset_log.c)
target_link_libraries(test_reset_log PRIVATE supervisor_core)
add_test(NAME reset_log_tests COMMAND test_reset_log)
