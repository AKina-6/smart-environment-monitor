cmake_minimum_required(VERSION 3.16)
project(dma_interrupt_buffer C)
set(CMAKE_C_STANDARD 11)
add_library(dma_core src/dma_io.c)
target_include_directories(dma_core PUBLIC include)
add_executable(dma_sim src/host_main.c)
target_link_libraries(dma_sim PRIVATE dma_core)
enable_testing()
add_executable(test_dma tests/test_dma.c)
target_link_libraries(test_dma PRIVATE dma_core)
add_test(NAME dma_tests COMMAND test_dma)
