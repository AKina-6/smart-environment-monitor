cmake_minimum_required(VERSION 3.16)
project(actuator_policy_hysteresis C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(actuator_core src/control/actuator_policy.c)
target_include_directories(actuator_core PUBLIC include)
add_executable(actuator_policy_sim src/platform/host/host_main.c)
target_link_libraries(actuator_policy_sim PRIVATE actuator_core)
enable_testing()
add_executable(test_policy tests/test_policy.c)
target_link_libraries(test_policy PRIVATE actuator_core)
add_test(NAME actuator_policy_tests COMMAND test_policy)
add_executable(test_fault tests/test_fault.c)
target_link_libraries(test_fault PRIVATE actuator_core)
add_test(NAME actuator_fault_tests COMMAND test_fault)
