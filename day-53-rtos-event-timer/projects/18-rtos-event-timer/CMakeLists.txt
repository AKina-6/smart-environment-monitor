cmake_minimum_required(VERSION 3.16)
project(rtos_event_timer C)
set(CMAKE_C_STANDARD 11)
add_library(rtos_event_timer src/event_timer.c)
target_include_directories(rtos_event_timer PUBLIC include)
add_executable(event_timer_sim src/host_main.c)
target_link_libraries(event_timer_sim PRIVATE rtos_event_timer)
enable_testing()
add_executable(test_event_timer tests/test_event_timer.c)
target_link_libraries(test_event_timer PRIVATE rtos_event_timer)
add_test(NAME event_timer_tests COMMAND test_event_timer)
