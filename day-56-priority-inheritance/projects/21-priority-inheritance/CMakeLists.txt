cmake_minimum_required(VERSION 3.16)
project(priority_inheritance C)
set(CMAKE_C_STANDARD 11)
add_library(priority_mutex src/priority_mutex.c)
target_include_directories(priority_mutex PUBLIC include)
add_executable(priority_sim src/host_main.c)
target_link_libraries(priority_sim PRIVATE priority_mutex)
enable_testing()
add_executable(test_priority tests/test_priority.c)
target_link_libraries(test_priority PRIVATE priority_mutex)
add_test(NAME priority_tests COMMAND test_priority)
