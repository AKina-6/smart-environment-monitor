cmake_minimum_required(VERSION 3.16)
project(device_driver_hal C)
set(CMAKE_C_STANDARD 11)
add_library(device_core src/device.c src/virtual_sensor.c src/hal_gpio.c)
target_include_directories(device_core PUBLIC include)
add_executable(device_sim src/host_main.c)
target_link_libraries(device_sim PRIVATE device_core)
enable_testing()
add_executable(test_device tests/test_device.c)
target_link_libraries(test_device PRIVATE device_core)
add_test(NAME device_tests COMMAND test_device)
