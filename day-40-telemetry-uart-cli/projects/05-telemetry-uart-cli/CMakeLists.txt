cmake_minimum_required(VERSION 3.16)
project(telemetry_uart_cli C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT MSVC)
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(cli_core src/cli/runtime_config.c src/cli/telemetry.c src/cli/cli.c)
target_include_directories(cli_core PUBLIC include)
add_executable(telemetry_cli_sim src/platform/host/host_main.c)
target_link_libraries(telemetry_cli_sim PRIVATE cli_core)
enable_testing()
add_executable(test_config tests/test_config.c)
target_link_libraries(test_config PRIVATE cli_core)
add_test(NAME config_tests COMMAND test_config)
add_executable(test_cli tests/test_cli.c)
target_link_libraries(test_cli PRIVATE cli_core)
add_test(NAME cli_tests COMMAND test_cli)
