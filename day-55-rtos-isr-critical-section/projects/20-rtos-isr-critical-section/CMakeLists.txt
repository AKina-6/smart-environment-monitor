cmake_minimum_required(VERSION 3.16)
project(rtos_isr_critical_section C)
set(CMAKE_C_STANDARD 11)
add_library(isr_sync src/isr_sync.c)
target_include_directories(isr_sync PUBLIC include)
add_executable(isr_sim src/host_main.c)
target_link_libraries(isr_sim PRIVATE isr_sync)
enable_testing()
add_executable(test_isr_sync tests/test_isr_sync.c)
target_link_libraries(test_isr_sync PRIVATE isr_sync)
add_test(NAME isr_sync_tests COMMAND test_isr_sync)
