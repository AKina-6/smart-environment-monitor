cmake_minimum_required(VERSION 3.16)
project(rtos_queue C)
set(CMAKE_C_STANDARD 11)
add_library(queue_core src/queue.c)
target_include_directories(queue_core PUBLIC include)
add_executable(queue_sim src/host_main.c)
target_link_libraries(queue_sim PRIVATE queue_core)
enable_testing()
add_executable(test_queue tests/test_queue.c)
target_link_libraries(test_queue PRIVATE queue_core)
add_test(NAME queue_tests COMMAND test_queue)
